cmake_minimum_required(VERSION 3.16)
project(VocalHiFi VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(vocalhifi SHARED vocal_hifi.cpp)
target_include_directories(vocalhifi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(vocal_hifi_selftest vocal_hifi_selftest.cpp)
enable_testing()
add_test(NAME vocal_hifi_selftest COMMAND vocal_hifi_selftest)
